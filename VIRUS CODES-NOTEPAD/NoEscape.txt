        public void final_payload()

            Process.Start(@"C:\Windows\notepad.exe");
            Process[] proces_name = Process.GetProcessesByName("notepad");
            Process[] proces_name2 = Process.GetProcessesByName("explorer");

            dport.INPUT[] input = new dport.INPUT[2];
            int x = Screen.PrimaryScreen.Bounds.Width; int y = Screen.PrimaryScreen.Bounds.Height;
            for (int num = 0; num < 150; num++)
            {                
                IntPtr find_win = dport.FindWindow("Notepad", null);
                dport.MoveWindow(find_win, 100, 100, x / 2, y / 3, true);

                UInt16 unicode = word_char[num];

                input[0].type = dport.InputType.INPUT_KEYBOARD;
                input[0].U.ki.wScan = (dport.ScanCodeShort)unicode;
                input[0].U.ki.dwFlags = dport.KEYEVENTF.UNICODE;

                input[1].type = dport.InputType.INPUT_KEYBOARD;
                input[1].U.ki.wVk = dport.VirtualKeyShort.RETURN;

                if(word_char[num] != ',' && word_char[num] != '.' && num != 129)

                else
                    dport.SendInput(2, input, Marshal.SizeOf(typeof(dport.INPUT)));
                Thread.Sleep(200);
            }

                IntPtr find_win = dport.FindWindow("Notepad", null);

                UInt16 unicode = word_char[num];

                input[0].type = dport.InputType.INPUT_KEYBOARD;
                input[0].U.ki.wScan = (dport.ScanCodeShort)unicode;
                input[0].U.ki.dwFlags = dport.KEYEVENTF.UNICODE;

                dport.SendInput(1, input, Marshal.SizeOf(typeof(dport.INPUT)));
